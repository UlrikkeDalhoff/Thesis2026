R 

library(data.table) 

 

expr <- fread("human_permissive_enhancers_phase_1_and_2_expression_tpm_matrix.txt") 

 

hela_cols <- c("CNhs12325","CNhs12326","CNhs12327") 

 

hela_expr <- expr[, c("Id", hela_cols), with = FALSE] 

 

#TPM threshold  

hela_active <- hela_expr[rowMeans(hela_expr[, ..hela_cols]) > 1] 

 

#get coordinates 

enh <- fread("human_permissive_enhancers_phase_1_and_2.bed") 

 

hela_enh <- enh[enh$V4 %in% hela_active$Id] 

 

#Save 

fwrite( 

 hela_enh[,1:3], 

 "HeLa_FANTOM5_enhancers.bed", 

 sep="\t", 

 col.names=FALSE 

)
